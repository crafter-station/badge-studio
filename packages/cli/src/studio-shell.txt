<script>
const site=__STUDIO_ORIGIN__;
const token=location.hash.slice(1);
const frame=document.querySelector("iframe");
const status=document.querySelector("#status");
const client=crypto.randomUUID();
let tools;
let ready=false;
const post=async(path,body)=>{
 const response=await fetch(path,{method:"POST",headers:{Authorization:`Bearer ${token}`,"Content-Type":"application/json"},body:JSON.stringify({client,...body})});
 if(!response.ok)throw new Error("Connection lost");
};
const forward=(message)=>frame.contentWindow.postMessage({channel:"badgio",token,...message},site);
if(!/^[a-f0-9]{64}$/.test(token)){
 status.textContent="Open the complete session URL from your agent.";
}else{
 const events=new EventSource(`/events?token=${token}&client=${client}`);
 const announce=async()=>{
  if(!ready||!tools)return;
  try{await post("/ready",{tools});status.textContent="Connected · Keep this tab open";}
  catch{status.textContent="Disconnected · Reopen the session URL";}
 };
 events.onopen=()=>{ready=true;void announce();forward({type:"discover"});};
 events.onmessage=(event)=>{
  const message=JSON.parse(event.data);
  if(message.type==="closed"){ready=false;events.close();forward({type:"cancel-all"});status.textContent="Session ended · You can still edit and export this badge";}
  else forward(message);
 };
 events.onerror=()=>{ready=false;forward({type:"cancel-all"});status.textContent="Disconnected · Close other preview tabs, then reconnect";};
 window.addEventListener("message",async(event)=>{
  const message=event.data;
  if(event.source!==frame.contentWindow||event.origin!==site||message?.channel!=="badgio"||message.token!==token)return;
  if(message.type==="ready"){tools=message.tools;await announce();}
  if(message.type==="result"){
   try{await post("/result",{id:message.id,output:message.output});}
   catch{status.textContent="Disconnected · Ask your agent to inspect before retrying";}
  }
 });
 window.addEventListener("pagehide",()=>{forward({type:"cancel-all"});events.close();});
 frame.src=`${site}/design#badgio=${token}&parent=${encodeURIComponent(location.origin)}`;
 setTimeout(()=>{if(!tools)status.textContent="Still connecting · Check that the website supports this CLI version";},15000);
}
</script>
